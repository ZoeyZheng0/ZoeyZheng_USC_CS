\subsection*{iLab, University of Southern California \hfill Los Angeles, CA}
\subtext{\textit{Research Assistant, Advisor: Prof. \href{http://ilab.usc.edu/itti/} {Laurent Itti} }\hfill \textit{May 2024 --- Present}}
    \begin{zitemize}
        \item \textbf{Three-Step Nav: A Hierarchical Global–Local Planner for Zero-Shot Vision-and-Language Navigation}
            \begin{itemize}
                \item Proposed a hierarchical global-local framework that alternates between \textit{looking forward} to sketch global plans, \textit{looking now} for fine-grained local grounding, and \textit{looking backward} to audit execution history.
                \item Designed an adaptive judge module equipped with meta-skills to detect navigation drift and trigger self-correction without task-specific fine-tuning.
                \item Achieved state-of-the-art zero-shot performance on R2R-CE and RxR-CE datasets, reducing navigation error by 15\% and improving SPL by 12\% on R2R-CE validation splits.
                \item One first-author paper accepted at AISTATS 2026 \cite{Pap6}.
            \end{itemize}
            
        \end{zitemize}
